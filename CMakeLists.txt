cmake_minimum_required(VERSION 3.20)
project(QuickOpenNpp LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "QuickOpenNpp must be built on Windows using Visual Studio / Windows SDK")
endif()
add_library(QuickOpenNpp SHARED QuickOpenNpp.cpp)
target_compile_features(QuickOpenNpp PRIVATE cxx_std_17)
target_compile_definitions(QuickOpenNpp PRIVATE UNICODE _UNICODE)
target_link_libraries(QuickOpenNpp PRIVATE user32 gdi32 comctl32)
set_target_properties(QuickOpenNpp PROPERTIES PREFIX "" OUTPUT_NAME "QuickOpenNpp")
target_link_libraries(QuickOpenNpp PRIVATE uxtheme)
